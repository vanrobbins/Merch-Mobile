```dart
// lib/features/zone_manager/mannequin_lock_card.dart
import 'package:flutter/material.dart';
import '../../core/database/app_database.dart';
import '../../core/theme/app_theme.dart';
import '../../core/theme/design_tokens.dart';

/// Displays the Mannequin Lock outfit summary card.
/// Pass [slots], [products] and [colors] lists pre-fetched from DB.
class MannequinLockCard extends StatelessWidget {
  const MannequinLockCard({
    super.key,
    required this.outfitName,
    required this.slots,
    required this.products,
    required this.colors,
  });

  final String outfitName;
  final List<MannequinOutfitSlotsTableData> slots;
  final List<ProductsTableData> products;
  final List<BrandColorsTableData> colors;

  static const _slotOrder = [
    'head', 'torso', 'legs', 'feet', 'left_hand', 'right_hand', 'accessory_1', 'accessory_2',
  ];

  static const _slotLabels = {
    'head': 'Head',
    'torso': 'Torso',
    'legs': 'Legs',
    'feet': 'Feet',
    'left_hand': 'L.Hand',
    'right_hand': 'R.Hand',
    'accessory_1': 'Acc 1',
    'accessory_2': 'Acc 2',
  };

  @override
  Widget build(BuildContext context) {
    final filledSlots = slots
        .where((s) => s.productId != null)
        .toList()
      ..sort((a, b) =>
          _slotOrder.indexOf(a.bodySlot).compareTo(_slotOrder.indexOf(b.bodySlot)));

    if (filledSlots.isEmpty) {
      return const Text(
        'No outfit',
        style: TextStyle(color: AppTheme.textSecondary, fontSize: DesignTokens.typeXs),
      );
    }

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      mainAxisSize: MainAxisSize.min,
      children: [
        Text(
          outfitName,
          style: const TextStyle(
            fontWeight: DesignTokens.weightBold,
            fontSize: DesignTokens.typeSm,
          ),
        ),
        const SizedBox(height: 4),
        ...filledSlots.map((slot) {
          final product = products.firstWhere(
            (p) => p.id == slot.productId,
            orElse: () => throw StateError('product not found'),
          );
          final color = slot.colorId != null
              ? colors.firstWhereOrNull((c) => c.id == slot.colorId)
              : null;
          final colorLabel = color?.name.toUpperCase() ??
              slot.colorNotes?.toUpperCase() ??
              '';
          return Padding(
            padding: const EdgeInsets.only(bottom: 2),
            child: Row(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                SizedBox(
                  width: 52,
                  child: Text(
                    '${_slotLabels[slot.bodySlot] ?? slot.bodySlot}:',
                    style: const TextStyle(
                      fontSize: DesignTokens.typeXs,
                      color: AppTheme.textSecondary,
                    ),
                  ),
                ),
                Expanded(
                  child: Text(
                    colorLabel.isNotEmpty
                        ? '${product.sku} — $colorLabel'
                        : product.sku,
                    style: const TextStyle(fontSize: DesignTokens.typeXs),
                  ),
                ),
              ],
            ),
          );
        }),
      ],
    );
  }
}
```

```dart
// lib/features/zone_manager/mannequin_dressing_sheet.dart
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_svg/flutter_svg.dart';
import 'package:uuid/uuid.dart';
import 'package:collection/collection.dart';

import '../../core/database/app_database.dart';
import '../../core/providers/auth_provider.dart';
import '../../core/providers/database_provider.dart';
import '../../core/providers/store_provider.dart';
import '../../core/theme/app_theme.dart';
import '../../core/theme/design_tokens.dart';

/// Available body slots per mannequin type.
const _slotsByType = {
  'full_body': ['head', 'torso', 'legs', 'feet', 'left_hand', 'right_hand', 'accessory_1', 'accessory_2'],
  'half_body': ['torso', 'left_hand', 'right_hand', 'accessory_1', 'accessory_2'],
  'torso': ['torso', 'accessory_1', 'accessory_2'],
  'leg_form': ['legs', 'feet'],
  'bra_form': ['torso'],
};

/// Categories of products assignable per slot.
const _slotCategories = {
  'head': ['Accessories'],
  'torso': ['Tops', 'Outerwear'],
  'legs': ['Bottoms'],
  'feet': ['Accessories'],
  'left_hand': ['Accessories'],
  'right_hand': ['Accessories'],

};

const _slotLabels = {

  'torso': 'Torso',
  'legs': 'Legs',
  'feet': 'Feet',
  'left_hand': 'Left Hand',
  'right_hand': 'Right Hand',
  'accessory_1': 'Accessory 1',
  'accessory_2': 'Accessory 2',
};

/// Opens the dressing sheet as a full-screen bottom sheet.
void showMannequinDressingSheet(BuildContext context, MannequinsTableData mannequin) {
  showModalBottomSheet<void>(
    context: context,
    isScrollControlled: true,
    backgroundColor: Colors.transparent,
    builder: (_) => MannequinDressingSheet(mannequin: mannequin),
  );
}

class MannequinDressingSheet extends ConsumerStatefulWidget {
  const MannequinDressingSheet({super.key, required this.mannequin});
  final MannequinsTableData mannequin;

  @override
  ConsumerState<MannequinDressingSheet> createState() =>
      _MannequinDressingSheetState();
}

class _MannequinDressingSheetState extends ConsumerState<MannequinDressingSheet> {
  late final TextEditingController _outfitNameCtrl;
  late final TextEditingController _outfitNotesCtrl;
  bool _saving = false;

  // Working copy of slot assignments: bodySlot → productId?
  late Map<String, String?> _slotProductMap;

  @override
  void initState() {
    super.initState();
    _outfitNameCtrl = TextEditingController(
      text: widget.mannequin.outfitName ?? widget.mannequin.name ?? '',
    );
    _outfitNotesCtrl = TextEditingController(
      text: widget.mannequin.outfitNotes ?? '',
    );
    _slotProductMap = {
      for (final slot in _slotsByType[widget.mannequin.mannequinType] ?? []) slot: null,
    };
  }

  @override
  void dispose() {
    _outfitNameCtrl.dispose();
    _outfitNotesCtrl.dispose();
    super.dispose();
  }

  List<String> get _availableSlots =>
      _slotsByType[widget.mannequin.mannequinType] ?? [];

  Future<void> _loadExistingSlots(List<MannequinOutfitSlotsTableData> slots) async {
    for (final slot in slots) {
      if (_slotsByType[widget.mannequin.mannequinType]?.contains(slot.bodySlot) == true) {
        _slotProductMap[slot.bodySlot] = slot.productId;
      }
    }
  }

  bool _isCoordinatorOrManager() {
    final membership = ref.read(currentMembershipProvider).value;
    return membership?.role == 'coordinator' || membership?.role == 'manager';
  }

  Future<void> _save(List<MannequinOutfitSlotsTableData> existingSlots) async {
    setState(() => _saving = true);
    try {
      final db = ref.read(appDatabaseProvider);
      final storeId = ref.read(activeStoreIdProvider).value ?? '';
      final now = DateTime.now().millisecondsSinceEpoch;

      if (_isCoordinatorOrManager()) {
        // Direct save: delete existing slots, insert new ones
        await db.mannequinOutfitSlotsDao.deleteByMannequin(widget.mannequin.id);
        for (final entry in _slotProductMap.entries) {
          if (entry.value != null) {
            await db.mannequinOutfitSlotsDao.upsert(
              MannequinOutfitSlotsTableCompanion.insert(
                id: const Uuid().v4(),
                mannequinId: widget.mannequin.id,
                bodySlot: entry.key,
                productId: Value(entry.value),
                createdAt: now,
              ),
            );
          }
        }
        // Update mannequin outfit name/notes
        await db.mannequinsDao.upsert(MannequinsTableCompanion(
          id: Value(widget.mannequin.id),
          storeId: Value(widget.mannequin.storeId),
          zoneId: Value(widget.mannequin.zoneId),
          mannequinType: Value(widget.mannequin.mannequinType),
          mountType: Value(widget.mannequin.mountType),
          positionX: Value(widget.mannequin.positionX),
          positionY: Value(widget.mannequin.positionY),
          rotation: Value(widget.mannequin.rotation),
          createdAt: Value(widget.mannequin.createdAt),
          outfitName: Value(_outfitNameCtrl.text.trim().isEmpty
              ? null
              : _outfitNameCtrl.text.trim()),
          outfitNotes: Value(_outfitNotesCtrl.text.trim().isEmpty
              ? null
              : _outfitNotesCtrl.text.trim()),
        ));
      } else {
        // Staff: create proposal
        final user = ref.read(authStateProvider).value;
        if (user == null) return;
        final slotChangesJson = jsonEncode(
          _slotProductMap.entries
              .where((e) => e.value != null)
              .map((e) => {'slot': e.key, 'productId': e.value})
              .toList(),
        );
        await db.mannequinOutfitProposalsDao.upsert(
          MannequinOutfitProposalsTableCompanion.insert(
            id: const Uuid().v4(),
            mannequinId: widget.mannequin.id,
            storeId: storeId,
            proposedByUid: user.uid,
            proposedAt: now,
            status: 'pending',
            slotChanges: slotChangesJson,
            notes: Value(_outfitNotesCtrl.text.trim().isEmpty
                ? null
                : _outfitNotesCtrl.text.trim()),
          ),
        );
      }
      if (mounted) Navigator.pop(context);
    } finally {
      if (mounted) setState(() => _saving = false);
    }
  }

  Future<void> _pickProduct(String slot) async {
    final categories = _slotCategories[slot] ?? ['Other'];
    final db = ref.read(appDatabaseProvider);
    final storeId = ref.read(activeStoreIdProvider).value ?? '';
    final allProducts = await db.productsDao.getByStore(storeId);
    final filtered = allProducts
        .where((p) => categories.contains(p.category))
        .toList();

    if (!mounted) return;
    final selected = await showModalBottomSheet<ProductsTableData>(
      context: context,
      builder: (_) => _ProductSlotPicker(
        products: filtered,
        slotLabel: _slotLabels[slot] ?? slot,
      ),
    );
    if (selected != null) {
      setState(() => _slotProductMap[slot] = selected.id);
    }
  }

  @override
  Widget build(BuildContext context) {
    final storeId = ref.watch(activeStoreIdProvider).value ?? '';
    final db = ref.watch(appDatabaseProvider);
    final isStaff = !_isCoordinatorOrManager();

    return DraggableScrollableSheet(
      initialChildSize: 0.92,
      minChildSize: 0.5,
      maxChildSize: 0.95,
      builder: (context, scrollCtrl) {
        return Container(
          decoration: const BoxDecoration(
            color: AppTheme.cardSurface,
            borderRadius: BorderRadius.vertical(top: Radius.circular(16)),
          ),
          child: StreamBuilder<List<MannequinOutfitSlotsTableData>>(
            stream: db.mannequinOutfitSlotsDao.watchByMannequin(widget.mannequin.id),
            builder: (context, slotsSnap) {
              final existingSlots = slotsSnap.data ?? [];
              // Sync working map from DB on first load
              if (slotsSnap.connectionState == ConnectionState.active &&
                  slotsSnap.data != null) {
                for (final slot in existingSlots) {
                  if (_slotProductMap.containsKey(slot.bodySlot)) {
                    _slotProductMap[slot.bodySlot] = slot.productId;
                  }
                }
              }

              return StreamBuilder<List<ProductsTableData>>(
                stream: db.productsDao.watchByStore(storeId),
                builder: (context, productsSnap) {
                  final products = productsSnap.data ?? [];

                  return Column(
                    children: [
                      // Handle + header
                      Padding(
                        padding: const EdgeInsets.symmetric(vertical: DesignTokens.spaceSm),
                        child: Container(
                          width: 40,
                          height: 4,
                          decoration: BoxDecoration(
                            color: AppTheme.primary.withOpacity(0.2),
                            borderRadius: BorderRadius.circular(2),
                          ),
                        ),
                      ),
                      Padding(
                        padding: const EdgeInsets.symmetric(horizontal: DesignTokens.spaceMd),
                        child: Row(
                          children: [
                            Expanded(
                              child: TextField(
                                controller: _outfitNameCtrl,
                                decoration: const InputDecoration(
                                  labelText: 'Outfit name',
                                  border: UnderlineInputBorder(),
                                ),
                                style: const TextStyle(
                                  fontWeight: DesignTokens.weightBold,
                                  fontSize: DesignTokens.typeMd,
                                ),
                              ),
                            ),
                            const SizedBox(width: DesignTokens.spaceMd),
                            TextButton(
                              onPressed: _saving ? null : () => _save(existingSlots),
                              child: Text(
                                isStaff ? 'PROPOSE' : 'SAVE',
                                style: const TextStyle(color: AppTheme.accent),
                              ),
                            ),
                          ],
                        ),
                      ),
                      if (isStaff)
                        Padding(
                          padding: const EdgeInsets.symmetric(horizontal: DesignTokens.spaceMd),
                          child: TextField(
                            controller: _outfitNotesCtrl,
                            decoration: const InputDecoration(
                              labelText: 'Notes for reviewer',
                              border: UnderlineInputBorder(),
                            ),
                            maxLines: 2,
                          ),
                        ),
                      const Divider(),
                      Expanded(
                        child: Row(
                          crossAxisAlignment: CrossAxisAlignment.start,
                          children: [
                            // Left: silhouette diagram
                            SizedBox(
                              width: 120,
                              child: Padding(
                                padding: const EdgeInsets.all(DesignTokens.spaceMd),
                                child: SvgPicture.asset(
                                  'assets/silhouettes/${_silhouetteForType(widget.mannequin.mannequinType)}.svg',
                                  colorFilter: const ColorFilter.mode(
                                    AppTheme.textSecondary,
                                    BlendMode.srcIn,
                                  ),
                                ),
                              ),
                            ),
                            const VerticalDivider(width: 1),
                            // Right: slot list

                              child: ListView(
                                controller: scrollCtrl,
                                padding: const EdgeInsets.all(DesignTokens.spaceSm),
                                children: _availableSlots.map((slot) {
                                  final productId = _slotProductMap[slot];
                                  final product = productId != null
                                      ? products.firstWhereOrNull((p) => p.id == productId)

                                  return _SlotRow(
                                    slotLabel: _slotLabels[slot] ?? slot,
```

```dart
// lib/features/zone_manager/outfit_proposal_review_screen.dart
import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:drift/drift.dart' show Value;
import '../../core/database/app_database.dart';
import '../../core/providers/auth_provider.dart';
import '../../core/providers/database_provider.dart';
import '../../core/providers/store_provider.dart';
import '../../core/theme/app_theme.dart';
import '../../core/theme/design_tokens.dart';

class OutfitProposalReviewScreen extends ConsumerWidget {
  const OutfitProposalReviewScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final storeId = ref.watch(activeStoreIdProvider).value ?? '';
    final db = ref.watch(appDatabaseProvider);

    return Scaffold(
      appBar: AppBar(title: const Text('OUTFIT PROPOSALS')),
      body: StreamBuilder<List<MannequinOutfitProposalsTableData>>(
        stream: db.mannequinOutfitProposalsDao.watchPendingByStore(storeId),
        builder: (context, snap) {
          final proposals = snap.data ?? [];
          if (proposals.isEmpty) {
            return const Center(
              child: Text(
                'No pending outfit proposals.',
                style: TextStyle(color: AppTheme.textSecondary),
              ),
            );
          }
          return ListView.builder(
            itemCount: proposals.length,
            itemBuilder: (_, i) => _ProposalCard(
              proposal: proposals[i],
              onApprove: () => _reviewProposal(context, ref, proposals[i], 'approved'),
              onReject: () => _reviewProposal(context, ref, proposals[i], 'rejected'),
            ),
          );
        },
      ),
    );
  }

  Future<void> _reviewProposal(
    BuildContext context,
    WidgetRef ref,
    MannequinOutfitProposalsTableData proposal,
    String decision,
  ) async {
    final db = ref.read(appDatabaseProvider);
    final user = ref.read(authStateProvider).value;
    final now = DateTime.now().millisecondsSinceEpoch;

    await db.mannequinOutfitProposalsDao.upsert(
      MannequinOutfitProposalsTableCompanion(
        id: Value(proposal.id),
        mannequinId: Value(proposal.mannequinId),
        storeId: Value(proposal.storeId),
        proposedByUid: Value(proposal.proposedByUid),
        proposedAt: Value(proposal.proposedAt),
        slotChanges: Value(proposal.slotChanges),
        status: Value(decision),
        reviewedByUid: Value(user?.uid),
        reviewedAt: Value(now),
      ),
    );

    if (decision == 'approved') {
      // Apply slot changes from the proposal
      final changes = (jsonDecode(proposal.slotChanges) as List)
          .cast<Map<String, dynamic>>();
      await db.mannequinOutfitSlotsDao.deleteByMannequin(proposal.mannequinId);
      for (final change in changes) {
        await db.mannequinOutfitSlotsDao.upsert(
          MannequinOutfitSlotsTableCompanion.insert(
            id: const Uuid().v4(),
            mannequinId: proposal.mannequinId,
            bodySlot: change['slot'] as String,
            productId: Value(change['productId'] as String?),
            createdAt: now,
          ),
        );
      }
      if (context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Outfit proposal approved and applied.')),
        );
      }
    } else {
      if (context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Outfit proposal rejected.')),
        );
      }
    }
  }
}

class _ProposalCard extends StatelessWidget {
  const _ProposalCard({
    required this.proposal,
    required this.onApprove,
    required this.onReject,

      changes = jsonDecode(proposal.slotChanges) as List;

                    fontSize: DesignTokens.typeSm,
                  ),
                ),
                Text(
                  dateStr,
                  style: const TextStyle(
                    color: AppTheme.textSecondary,
                    fontSize: DesignTokens.typeXs,
                  ),
                ),
              ],
            ),
            if (proposal.notes != null && proposal.notes!.isNotEmpty) ...[
              const SizedBox(height: 4),

                '"${proposal.notes}"',
                style: const TextStyle(
                  color: AppTheme.textSecondary,
                  fontStyle: FontStyle.italic,
                  fontSize: DesignTokens.typeXs,
                ),
              ),
            ],
            const SizedBox(height: DesignTokens.spaceSm),
            Text(
              '${changes.length} slot(s) proposed',
              style: const TextStyle(fontSize: DesignTokens.typeXs),
            ),
            const SizedBox(height: DesignTokens.spaceMd),
            Row(
              mainAxisAlignment: MainAxisAlignment.end,
              children: [
                OutlinedButton(
                  onPressed: onReject,
                  style: OutlinedButton.styleFrom(foregroundColor: Colors.red),
                  child: const Text('REJECT'),
                ),
                const SizedBox(width: DesignTokens.spaceSm),
                ElevatedButton(
                  onPressed: onApprove,
                  style: ElevatedButton.styleFrom(backgroundColor: AppTheme.accent),
                  child: const Text('APPROVE'),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }
}
```
