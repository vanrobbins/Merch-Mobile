      id: 'c1',
      storeId: 's1',
      name: 'Navy',
      hexValue: '#1B2A4A',
      createdAt: 1000,
    ));
    await db.brandColorsDao.deleteById('c1');

// test/integration/mannequin_dressing_flow_test.dart
import 'package:drift/native.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:merch_mobile/core/database/app_database.dart';
import 'package:merch_mobile/core/services/seed_service.dart';

void main() {
  late AppDatabase db;

  setUp(() => db = AppDatabase.forTesting(NativeDatabase.memory()));
  tearDown(() async => db.close());

  test('full flow: seed templates → create color → create product → place mannequin → dress mannequin', () async {
    const storeId = 'test_store';

    // 1. Seed global templates
    await seedGlobalTemplates(db);
    final templates = await (db.productTemplatesDao.watchBuiltIn().first);
    expect(templates.length, 13);

    // 2. Create brand color
    await db.brandColorsDao.upsert(BrandColorsTableCompanion.insert(
      id: 'c1', storeId: storeId, name: 'Navy', hexValue: '#1B2A4A', createdAt: 1000,
    ));
    final colors = await db.brandColorsDao.getByStore(storeId);
    expect(colors.length, 1);

    // 3. Create product using template + color
    final shirtTemplate = templates.firstWhere((t) => t.silhouetteType == 'shirt_ss');
    await db.productsDao.upsert(ProductsTableCompanion.insert(
      id: 'prod1',
      sku: 'MVS-NV01',
      name: 'Metal Vent SS Navy',
      category: 'Tops',
      updatedAt: DateTime.fromMillisecondsSinceEpoch(1000),
      storeId: const Value(storeId),
      templateId: Value(shirtTemplate.id),
      colorId: const Value('c1'),
    ));
    final products = await db.productsDao.getByStore(storeId);
    expect(products.length, 1);
    expect(products.first.colorId, 'c1');

    // 4. Place mannequin in a zone
    await db.mannequinsDao.upsert(MannequinsTableCompanion.insert(
      id: 'm1',
      storeId: storeId,
      zoneId: 'z1',
      mannequinType: 'full_body',
      mountType: 'floor',
      positionX: 5.0,
      positionY: 5.0,
      createdAt: 2000,
    ));
    final mannequin = await db.mannequinsDao.findById('m1');
    expect(mannequin, isNotNull);

    // 5. Dress mannequin (coordinator path: direct slot save)
    await db.mannequinOutfitSlotsDao.upsert(
      MannequinOutfitSlotsTableCompanion.insert(
        id: 'slot1',
        mannequinId: 'm1',
        bodySlot: 'torso',
        productId: const Value('prod1'),
        colorId: const Value('c1'),
        createdAt: 3000,
      ),
    );
    final slots = await db.mannequinOutfitSlotsDao.getByMannequin('m1');
    expect(slots.length, 1);
    expect(slots.first.bodySlot, 'torso');
    expect(slots.first.productId, 'prod1');

    // 6. Verify MannequinLockCard data available
    final filledSlots = slots.where((s) => s.productId != null).toList();
    expect(filledSlots.length, 1);
    expect(filledSlots.first.productId, 'prod1');
  });

  test('staff proposal flow: create proposal → approve → slots applied', () async {
    const storeId = 'test_store';

    // Setup: place mannequin
    await db.mannequinsDao.upsert(MannequinsTableCompanion.insert(
      id: 'm1', storeId: storeId, zoneId: 'z1',
      mannequinType: 'torso', mountType: 'wall',
      positionX: 2.0, positionY: 2.0, createdAt: 1000,
    ));

    // Create a product
    await db.productsDao.upsert(ProductsTableCompanion.insert(
      id: 'prod1', sku: 'TK-GY01', name: 'Tank Grey', category: 'Tops',
      updatedAt: DateTime.fromMillisecondsSinceEpoch(1000),
      storeId: const Value(storeId),
    ));

    // Staff submits a proposal
    await db.mannequinOutfitProposalsDao.upsert(
      MannequinOutfitProposalsTableCompanion.insert(
        id: 'prop1',
        mannequinId: 'm1',
        storeId: storeId,
        proposedByUid: 'staff_uid',
        proposedAt: 2000,
        status: 'pending',
        slotChanges: '[{"slot":"torso","productId":"prod1"}]',
      ),
    );

    final pending = await (db.mannequinOutfitProposalsDao.watchPendingByStore(storeId).first);
    expect(pending.length, 1);

    // Manager approves
    await db.mannequinOutfitProposalsDao.upsert(
      MannequinOutfitProposalsTableCompanion(
        id: const Value('prop1'),
        mannequinId: const Value('m1'),
        storeId: const Value(storeId),
        proposedByUid: const Value('staff_uid'),
        proposedAt: const Value(2000),
        slotChanges: const Value('[{"slot":"torso","productId":"prod1"}]'),
        status: const Value('approved'),
        reviewedByUid: const Value('manager_uid'),
        reviewedAt: const Value(3000),
      ),
    );
    // Apply slot changes
    await db.mannequinOutfitSlotsDao.upsert(
      MannequinOutfitSlotsTableCompanion.insert(
        id: 'slot1', mannequinId: 'm1', bodySlot: 'torso',
        productId: const Value('prod1'), createdAt: 3000,
      ),
    );

    final stillPending = await (db.mannequinOutfitProposalsDao.watchPendingByStore(storeId).first);
    expect(stillPending, isEmpty);

    final slots = await db.mannequinOutfitSlotsDao.getByMannequin('m1');
    expect(slots.length, 1);
    expect(slots.first.productId, 'prod1');
  });
}
