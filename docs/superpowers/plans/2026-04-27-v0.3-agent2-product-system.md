// lib/core/widgets/mm_color_picker.dart
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import '../../core/database/app_database.dart';
import '../../core/theme/app_theme.dart';

/// Displays brand palette chips + an "Other" chip that opens a hex input.
/// [colors] — list of BrandColorsTableData from store palette.
/// [selectedColorId] — currently selected palette color id (null = none / other).
/// [colorNotes] — free text when "Other" is selected.
/// [onColorSelected] — called with (colorId, colorNotes):
///   colorId non-null + colorNotes null = palette color selected
///   colorId null + colorNotes non-null = "other" hex entered
///   both null = cleared
class MmColorPicker extends StatefulWidget {
  const MmColorPicker({
    super.key,
    required this.colors,
    this.selectedColorId,
    this.colorNotes,
    required this.onColorSelected,
  });

  final List<BrandColorsTableData> colors;
  final String? selectedColorId;
  final String? colorNotes;
  final void Function(String? colorId, String? colorNotes) onColorSelected;

  @override
  State<MmColorPicker> createState() => _MmColorPickerState();
}

class _MmColorPickerState extends State<MmColorPicker> {
  bool _showOther = false;
  final _hexCtrl = TextEditingController();

  @override
  void initState() {
    super.initState();
    if (widget.selectedColorId == null && widget.colorNotes != null) {
      _showOther = true;
      _hexCtrl.text = widget.colorNotes ?? '';
    }
  }

  @override
  void dispose() {
    _hexCtrl.dispose();
    super.dispose();
  }

  Color _parseHex(String hex) {
    final cleaned = hex.replaceAll('#', '');
    if (cleaned.length == 6) {
      return Color(int.parse('FF$cleaned', radix: 16));
    }
    return Colors.grey;
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Wrap(
          spacing: DesignTokens.spaceSm,
          runSpacing: DesignTokens.spaceSm,
          children: [
            ...widget.colors.map((c) {
              final isSelected = c.id == widget.selectedColorId;
              final chipColor = _parseHex(c.hexValue);
              return GestureDetector(
                onTap: () {
                  setState(() => _showOther = false);
                  widget.onColorSelected(c.id, null);
                },
                child: AnimatedContainer(
                  duration: const Duration(milliseconds: 150),
                  padding: const EdgeInsets.symmetric(
                    horizontal: DesignTokens.spaceSm,
                    vertical: 4,
                  ),
                  decoration: BoxDecoration(
                    color: isSelected
                        ? AppTheme.primary.withOpacity(0.08)
                        : Colors.transparent,
                    border: Border.all(
                      color: isSelected ? AppTheme.accent : AppTheme.primary.withOpacity(0.2),
                      width: isSelected ? 2 : 1,
                    ),
                    borderRadius: BorderRadius.circular(AppTheme.borderRadius),
                  ),
                  child: Row(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      Container(
                        width: 16,
                        height: 16,
                        decoration: BoxDecoration(
                          color: chipColor,
                          shape: BoxShape.circle,
                          border: Border.all(
                            color: AppTheme.primary.withOpacity(0.3),
                          ),
                        ),
                      ),
                      const SizedBox(width: 6),
                      Text(
                        c.name,
                        style: TextStyle(
                          fontSize: DesignTokens.typeSm,
                          fontWeight: isSelected ? DesignTokens.weightBold : FontWeight.normal,
                          color: AppTheme.textPrimary,
                        ),
                      ),
                    ],
                  ),
                ),
              );
            }),
            // "Other" chip
            GestureDetector(
              onTap: () {
                setState(() => _showOther = true);
                widget.onColorSelected(null, _hexCtrl.text.isEmpty ? null : _hexCtrl.text);
              },
              child: AnimatedContainer(
                duration: const Duration(milliseconds: 150),
                padding: const EdgeInsets.symmetric(
                  horizontal: DesignTokens.spaceSm,
                  vertical: 4,
                ),
                decoration: BoxDecoration(
                  color: _showOther
                      ? AppTheme.primary.withOpacity(0.08)
                      : Colors.transparent,
                  border: Border.all(
                    color: _showOther ? AppTheme.accent : AppTheme.primary.withOpacity(0.2),
                    width: _showOther ? 2 : 1,
                  ),
                  borderRadius: BorderRadius.circular(AppTheme.borderRadius),
                ),
                child: Text(
                  'Other',
                  style: TextStyle(
                    fontSize: DesignTokens.typeSm,
                    color: AppTheme.textSecondary,
                  ),
                ),
              ),
            ),
          ],
        ),
        if (_showOther) ...[
          const SizedBox(height: DesignTokens.spaceSm),
          TextField(
            controller: _hexCtrl,
            decoration: const InputDecoration(
              labelText: 'Color description or hex (e.g. #4A6FA5)',
              prefixText: '',
            ),
            inputFormatters: [LengthLimitingTextInputFormatter(40)],
            onChanged: (v) => widget.onColorSelected(null, v.isEmpty ? null : v),
          ),
        ],
      ],
    );
  }
}

          child: const Icon(Icons.add, color: Colors.white),
        ),
      ),
    );
  }

  void _showAddSheet(BuildContext context, WidgetRef ref, String storeId) {
    showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      builder: (_) => _ColorEditSheet(
        storeId: storeId,
        existing: null,
        onSave: (name, hex) async {
          final db = ref.read(appDatabaseProvider);
          await db.brandColorsDao.upsert(BrandColorsTableCompanion.insert(
            id: const Uuid().v4(),
            storeId: storeId,
            name: name,
            hexValue: hex,
            createdAt: DateTime.now().millisecondsSinceEpoch,
          ));
          if (context.mounted) Navigator.pop(context);
        },
      ),
    );
  }

  void _showEditSheet(
    BuildContext context,
    WidgetRef ref,
    String storeId,
    BrandColorsTableData color,
  ) {
    showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      builder: (_) => _ColorEditSheet(
        storeId: storeId,
        existing: color,
        onSave: (name, hex) async {
          final db = ref.read(appDatabaseProvider);
          await db.brandColorsDao.upsert(BrandColorsTableCompanion.insert(
            id: color.id,
            storeId: storeId,
            name: name,
            hexValue: hex,
            createdAt: color.createdAt,
          ));
          if (context.mounted) Navigator.pop(context);
        },
      ),
    );
  }

  Future<void> _confirmDelete(
    BuildContext context,
    WidgetRef ref,
    BrandColorsTableData color,
  ) async {
    final confirm = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('Delete Color?'),
        content: Text(
          'Remove "${color.name}" from the palette?\n\nProducts using this color will retain their color_id reference.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('CANCEL'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('DELETE', style: TextStyle(color: Colors.red)),
          ),
        ],
      ),
    );
    if (confirm == true && context.mounted) {
      await ref.read(appDatabaseProvider).brandColorsDao.deleteById(color.id);
    }
  }
}

class _ColorEditSheet extends StatefulWidget {
  const _ColorEditSheet({
    required this.storeId,
    required this.existing,
    required this.onSave,
  });

  final String storeId;
  final BrandColorsTableData? existing;
  final Future<void> Function(String name, String hex) onSave;

  State<_ColorEditSheet> createState() => _ColorEditSheetState();

class _ColorEditSheetState extends State<_ColorEditSheet> {
  late final TextEditingController _nameCtrl;
  late final TextEditingController _hexCtrl;
  bool _saving = false;

  @override
  void initState() {
    super.initState();
    _nameCtrl = TextEditingController(text: widget.existing?.name ?? '');
    _hexCtrl = TextEditingController(text: widget.existing?.hexValue ?? '#');
  }

  @override
  void dispose() {
    _nameCtrl.dispose();
    _hexCtrl.dispose();
    super.dispose();
  }

  Future<void> _save() async {
    final name = _nameCtrl.text.trim();
    final hex = _hexCtrl.text.trim();
    if (name.isEmpty || hex.isEmpty) return;
    setState(() => _saving = true);
    try {
      await widget.onSave(name, hex);
    } finally {
      if (mounted) setState(() => _saving = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: EdgeInsets.only(
        left: DesignTokens.spaceLg,
        right: DesignTokens.spaceLg,
        top: DesignTokens.spaceLg,
        bottom: MediaQuery.of(context).viewInsets.bottom + DesignTokens.spaceLg,
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.stretch,
        children: [
          Text(
            widget.existing == null ? 'ADD COLOR' : 'EDIT COLOR',
            style: const TextStyle(
              fontSize: DesignTokens.typeMd,
              fontWeight: DesignTokens.weightBold,
              letterSpacing: DesignTokens.letterSpacingEyebrow,

            controller: _hexCtrl,

          ),
          const SizedBox(height: DesignTokens.spaceLg),
          ElevatedButton(
            onPressed: _saving ? null : _save,
            style: ElevatedButton.styleFrom(backgroundColor: AppTheme.accent),
            child: _saving
                ? const CircularProgressIndicator(color: Colors.white)
                : Text(widget.existing == null ? 'ADD COLOR' : 'SAVE'),
          ),
        ],
      ),
    );
  }
}

class ProductFormScreen extends ConsumerStatefulWidget {
  const ProductFormScreen({super.key, this.productId});
  final String? productId;

  @override
  ConsumerState<ProductFormScreen> createState() => _ProductFormScreenState();
}

class _ProductFormScreenState extends ConsumerState<ProductFormScreen> {
  int _step = 0; // 0=template, 1=color, 2=details
  bool _loading = false;
  ProductsTableData? _existing;

  // Step 1: template
  String? _selectedTemplateId;
  String? _selectedSilhouetteType;
  String? _selectedTemplateCategory;

  // Step 2: color
  String? _selectedColorId;
  String? _colorNotes;

  // Step 3: details
  final _nameCtrl = TextEditingController();
  final _skuCtrl = TextEditingController();
  final _stockCtrl = TextEditingController(text: '0');
  final _priceCtrl = TextEditingController();

  @override
  void initState() {
    super.initState();
    if (widget.productId != null) _loadExisting();
  }

  Future<void> _loadExisting() async {
    final db = ref.read(appDatabaseProvider);
    final product = await db.productsDao.findById(widget.productId!);
    if (product != null && mounted) {
      setState(() {
        _existing = product;
        _selectedTemplateId = product.templateId;
        _selectedColorId = product.colorId;
        _colorNotes = product.colorNotes;
        _nameCtrl.text = product.name;
        _skuCtrl.text = product.sku;
        _stockCtrl.text = product.stockQty.toString();
        _priceCtrl.text = product.price?.toStringAsFixed(2) ?? '';
        _step = 2; // jump to details for edit
      });
    }
  }

  @override
  void dispose() {
    _nameCtrl.dispose();
    _skuCtrl.dispose();
    _stockCtrl.dispose();
    _priceCtrl.dispose();
    super.dispose();
  }

  void _autoFillName(String templateName) {
    if (_nameCtrl.text.isEmpty) _nameCtrl.text = templateName;
  }

  void _autoFillSku() {
    if (_skuCtrl.text.isEmpty && _selectedTemplateId != null) {
      final templateCode = (_selectedSilhouetteType ?? 'PRD')
          .replaceAll('_', '')
          .toUpperCase()
          .substring(0, (_selectedSilhouetteType?.replaceAll('_','').length ?? 3).clamp(0, 3));
      final colorCode = _selectedColorId?.substring(_selectedColorId!.length - 4).toUpperCase() ?? '00';
      _skuCtrl.text = '$templateCode-$colorCode';
    }
  }

  Future<void> _save() async {
    final name = _nameCtrl.text.trim();
    final sku = _skuCtrl.text.trim();
    if (name.isEmpty || sku.isEmpty) return;

    setState(() => _loading = true);
    try {
      final db = ref.read(appDatabaseProvider);
      final storeId = ref.read(activeStoreIdProvider).value ?? '';
      await db.productsDao.upsert(ProductsTableCompanion.insert(
        id: _existing?.id ?? const Uuid().v4(),
        sku: sku,
        name: name,
        category: _selectedTemplateCategory ?? _existing?.category ?? 'Other',
        storeId: Value(storeId),
        stockQty: Value(int.tryParse(_stockCtrl.text) ?? 0),
        updatedAt: DateTime.now(),
        templateId: Value(_selectedTemplateId),
        colorId: Value(_selectedColorId),
        colorNotes: Value(_colorNotes),
        price: Value(double.tryParse(_priceCtrl.text)),
      ));
      if (mounted) Navigator.pop(context);
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _delete() async {
    if (_existing == null) return;
    final confirm = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('Delete Product?'),
        actions: [
          TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('CANCEL')),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('DELETE', style: TextStyle(color: Colors.red)),
          ),
        ],
      ),
    );
    if (confirm == true && mounted) {
      await ref.read(appDatabaseProvider).productsDao.deleteById(_existing!.id);
      if (mounted) Navigator.pop(context);
    }
  }

  @override
  Widget build(BuildContext context) {
    final storeId = ref.watch(activeStoreIdProvider).value ?? '';
    final db = ref.watch(appDatabaseProvider);

    return Scaffold(
      appBar: AppBar(
        title: Text(widget.productId == null ? 'ADD PRODUCT' : 'EDIT PRODUCT'),
        actions: [
          if (_existing != null)
            IconButton(
              icon: const Icon(Icons.delete_outline, color: Colors.red),
              onPressed: _delete,
            ),
        ],
      ),
      body: widget.productId != null && _existing == null && !_loading
          ? const Center(child: CircularProgressIndicator())
          : Column(
              children: [
                _StepIndicator(currentStep: _step),
                Expanded(
                  child: StreamBuilder<List<ProductTemplatesTableData>>(
                    stream: db.productTemplatesDao.watchAllForStore(storeId),
                    builder: (context, templateSnap) {
                      return StreamBuilder<List<BrandColorsTableData>>(
                        stream: db.brandColorsDao.watchByStore(storeId),
                        builder: (context, colorSnap) {
                          final templates = templateSnap.data ?? [];
                          final colors = colorSnap.data ?? [];
                          return _buildStep(templates, colors);
                        },
                      );
                    },
                  ),
                ),
              ],
            ),
    );
  }

  Widget _buildStep(
    List<ProductTemplatesTableData> templates,
    List<BrandColorsTableData> colors,
  ) {
    switch (_step) {
      case 0:
        return _TemplateStep(
          templates: templates,
          selectedId: _selectedTemplateId,
          onSelect: (t) {
            setState(() {
              _selectedTemplateId = t.id;
              _selectedSilhouetteType = t.silhouetteType;
              _selectedTemplateCategory = t.category;
            });
            _autoFillName(t.name);
          },
          onNext: _selectedTemplateId != null ? () => setState(() => _step = 1) : null,
        );
      case 1:
        return _ColorStep(
          colors: colors,
          selectedColorId: _selectedColorId,
          colorNotes: _colorNotes,
          onColorSelected: (colorId, notes) =>
              setState(() {
                _selectedColorId = colorId;
                _colorNotes = notes;
              }),
          onBack: () => setState(() => _step = 0),
          onNext: () {
            _autoFillSku();
            setState(() => _step = 2);
          },
        );
      case 2:
        return _DetailsStep(
          nameCtrl: _nameCtrl,
          skuCtrl: _skuCtrl,
          stockCtrl: _stockCtrl,
          priceCtrl: _priceCtrl,
          selectedSilhouetteType: _selectedSilhouetteType,
          onBack: () => setState(() => _step = 1),
          onSave: _loading ? null : _save,
          isEdit: _existing != null,
        );
      default:
        return const SizedBox.shrink();
    }
  }
}

class _StepIndicator extends StatelessWidget {
  const _StepIndicator({required this.currentStep});
  final int currentStep;

  static const _labels = ['Template', 'Color', 'Details'];

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: const EdgeInsets.symmetric(
        horizontal: DesignTokens.spaceLg,
        vertical: DesignTokens.spaceMd,
      ),
      color: AppTheme.primary,
      child: Row(
        children: List.generate(_labels.length * 2 - 1, (i) {
          if (i.isOdd) {
            return Expanded(
              child: Container(
                height: 1,
                color: Colors.white24,
              ),
            );
          }
          final stepIndex = i ~/ 2;
          final isActive = stepIndex == currentStep;
          final isDone = stepIndex < currentStep;
          return Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Container(
                width: 28,
                height: 28,
                decoration: BoxDecoration(
                  color: isDone
                      ? AppTheme.accent
                      : isActive
                          ? Colors.white
                          : Colors.white24,
                  shape: BoxShape.circle,
                ),
                alignment: Alignment.center,
                child: isDone
                    ? const Icon(Icons.check, size: 16, color: Colors.white)
                    : Text(
                        '${stepIndex + 1}',
                        style: TextStyle(
                          color: isActive ? AppTheme.primary : Colors.white60,
                          fontWeight: DesignTokens.weightBold,
                          fontSize: DesignTokens.typeXs,
                        ),
                      ),
              ),
              const SizedBox(height: 4),
              Text(
                _labels[stepIndex],
                style: TextStyle(
                  color: isActive ? Colors.white : Colors.white54,
                  fontSize: 10,
                ),
              ),
            ],
          );
        }),
      ),
    );
  }
}
