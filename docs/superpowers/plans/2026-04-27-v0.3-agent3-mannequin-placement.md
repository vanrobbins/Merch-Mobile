```dart
Color _parseHex(String hex) {
  final cleaned = hex.replaceAll('#', '');
  if (cleaned.length == 6) {
    return Color(int.parse('FF$cleaned', radix: 16));
  }
  return const Color(0xFFCCCCCC);
}
```

```dart
// Layer 4: Mannequins (topmost)
for (final mannequin in mannequins) {
  final cx = mannequin.positionX * pixelsPerFt;
  final cy = mannequin.positionY * pixelsPerFt;
  final isSelected = mannequin.id == selectedMannequinId;
  final radius = 12.0;

  final fillPaint = Paint()
    ..style = PaintingStyle.fill
    ..color = const Color(0xFF888888);
  final borderPaint = Paint()
    ..style = PaintingStyle.stroke
    ..strokeWidth = isSelected ? 2.5 : 1.5
    ..color = isSelected ? const Color(0xFFBF5534) : const Color(0xFF555555);

  if (mannequin.mountType == 'wall') {
    // Half-circle for wall mount
    final rect = Rect.fromCircle(center: Offset(cx, cy), radius: radius);
    canvas.drawArc(rect, -pi / 2, pi, true, fillPaint);
    canvas.drawArc(rect, -pi / 2, pi, false, borderPaint);
  } else {
    canvas.drawCircle(Offset(cx, cy), radius, fillPaint);
    canvas.drawCircle(Offset(cx, cy), radius, borderPaint);
  }

  // Type badge
  final badge = _mannequinBadge(mannequin.mannequinType);
  _drawLabel(canvas, badge, Offset(cx, cy), size: 8, color: Colors.white);
}
```

- [ ] **Step 1: Add wall snap logic to _onPointerMove**

In `_FloorBuilderScreenState`, add a helper that checks if a mannequin should snap to a wall:

```dart
static const _wallSnapThresholdPx = 20.0;

/// Returns the snapped position and rotation if within threshold of any zone edge.
/// Only applies to wall-mount mannequin types.
({Offset snappedFt, double rotation})? _tryWallSnap(
  Offset positionFt,
  List<ZoneEdge> edges,
) {
  final positionPx = positionFt * _pixelsPerFt;
  ZoneEdge? nearest;
  double nearestDist = double.infinity;
  for (final edge in edges) {
    // Distance from point to line segment
    final dx = edge.endPx.dx - edge.startPx.dx;
    final dy = edge.endPx.dy - edge.startPx.dy;
    final lenSq = dx * dx + dy * dy;
    if (lenSq == 0) continue;
    final t = ((positionPx.dx - edge.startPx.dx) * dx +
            (positionPx.dy - edge.startPx.dy) * dy) /
        lenSq;
    final clamped = t.clamp(0.0, 1.0);
    final projX = edge.startPx.dx + clamped * dx;
    final projY = edge.startPx.dy + clamped * dy;
    final dist = (positionPx - Offset(projX, projY)).distance;
    if (dist < nearestDist) {
      nearestDist = dist;
      nearest = edge;
    }
  }
  if (nearest == null || nearestDist > _wallSnapThresholdPx) return null;
  final snappedPx = nearest.midPx;
  return (
    snappedFt: snappedPx / _pixelsPerFt,
    rotation: nearest.angleDeg + 90,
  );
}
```

Note: `ZoneEdge` is in `zone_edge_helper.dart`. `Offset` doesn't support `/ scalar` directly — use `Offset(snappedPx.dx / _pixelsPerFt, snappedPx.dy / _pixelsPerFt)` instead.

- [ ] **Step 2: Create zone_overlap_checker.dart**

```dart
import 'dart:math';
import 'package:flutter/foundation.dart';

class ZoneOverlapChecker {
  /// Returns true if convex polygons [a] and [b] overlap using SAT.
  static bool polygonsOverlap(List<Offset> a, List<Offset> b) {
    for (final poly in [a, b]) {
      for (int i = 0; i < poly.length; i++) {
        final p1 = poly[i];
        final p2 = poly[(i + 1) % poly.length];
        final edge = p2 - p1;
        final axis = Offset(-edge.dy, edge.dx); // perpendicular

        double minA = double.infinity, maxA = double.negativeInfinity;
        double minB = double.infinity, maxB = double.negativeInfinity;

        for (final v in a) {
          final proj = v.dx * axis.dx + v.dy * axis.dy;
          if (proj < minA) minA = proj;
          if (proj > maxA) maxA = proj;
        }
        for (final v in b) {
          final proj = v.dx * axis.dx + v.dy * axis.dy;
          if (proj < minB) minB = proj;
          if (proj > maxB) maxB = proj;
        }

        if (maxA < minB || maxB < minA) return false; // separating axis found
      }
    }
    return true;
  }

  /// Returns IDs of all zones that overlap with at least one other zone.
  static Set<String> findOverlappingZoneIds(Map<String, List<Offset>> zones) {
    final ids = zones.keys.toList();
    final overlapping = <String>{};
    for (int i = 0; i < ids.length; i++) {
      for (int j = i + 1; j < ids.length; j++) {
        if (polygonsOverlap(zones[ids[i]]!, zones[ids[j]]!)) {
          overlapping.add(ids[i]);
          overlapping.add(ids[j]);
        }
      }
    }
    return overlapping;
  }
}
```
