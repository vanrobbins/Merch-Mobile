```dart
// lib/core/database/daos/brand_colors_dao.dart
import 'package:drift/drift.dart';
import '../app_database.dart';
import '../tables/brand_colors_table.dart';

part 'brand_colors_dao.g.dart';

@DriftAccessor(tables: [BrandColorsTable])
class BrandColorsDao extends DatabaseAccessor<AppDatabase>
    with _$BrandColorsDaoMixin {
  BrandColorsDao(super.db);

  Stream<List<BrandColorsTableData>> watchByStore(String storeId) =>
      (select(brandColorsTable)
            ..where((t) => t.storeId.equals(storeId))
            ..orderBy([(t) => OrderingTerm.asc(t.name)]))
          .watch();

  Future<List<BrandColorsTableData>> getByStore(String storeId) =>
      (select(brandColorsTable)
            ..where((t) => t.storeId.equals(storeId)))
          .get();

  Future<BrandColorsTableData?> findById(String id) =>
      (select(brandColorsTable)..where((t) => t.id.equals(id)))
          .getSingleOrNull();

  Future<void> upsert(BrandColorsTableCompanion row) =>
      into(brandColorsTable).insertOnConflictUpdate(row);

  Future<void> deleteById(String id) =>
      (delete(brandColorsTable)..where((t) => t.id.equals(id))).go();
}
```

```dart
                t.storeId.equals(storeId) & t.zoneId.equals(zoneId)))
          .watch();

  Future<void> upsert(ScenePropsTableCompanion row) =>
      into(scenePropsTable).insertOnConflictUpdate(row);

  Future<void> deleteById(String id) =>
      (delete(scenePropsTable)..where((t) => t.id.equals(id))).go();
}
```
