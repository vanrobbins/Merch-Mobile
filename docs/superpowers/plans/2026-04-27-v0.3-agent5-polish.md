### Task 2: Refined mannequin canvas rendering

**Files:**
- Modify: `lib/features/floor_builder/builder_canvas_painter.dart`

- [ ] **Step 1: Improve mannequin body icon**

Replace the simple circle mannequin rendering with a stylized figure. In the mannequin rendering loop, replace the `canvas.drawCircle` calls with a small "stick figure" shape:

```dart
// Layer 4: Mannequins — stylized figure
for (final mannequin in mannequins) {
  final cx = mannequin.positionX * pixelsPerFt;
  final cy = mannequin.positionY * pixelsPerFt;
  final isSelected = mannequin.id == selectedMannequinId;

  final fillColor = const Color(0xFF888888);
  final borderColor = isSelected ? const Color(0xFFBF5534) : const Color(0xFF555555);
  final fillPaint = Paint()..style = PaintingStyle.fill..color = fillColor;
  final strokePaint = Paint()
    ..style = PaintingStyle.stroke
    ..strokeWidth = isSelected ? 2.5 : 1.5
    ..color = borderColor;

  if (mannequin.mountType == 'wall') {
    // Half-circle flush to wall edge, facing inward
    final rect = Rect.fromCircle(center: Offset(cx, cy), radius: 12.0);
    canvas.drawArc(rect, -pi / 2, pi, true, fillPaint);
    canvas.drawArc(rect, -pi / 2, pi, false, strokePaint);
  } else {
    // Head
    canvas.drawCircle(Offset(cx, cy - 10), 5, fillPaint);
    canvas.drawCircle(Offset(cx, cy - 10), 5, strokePaint);
    // Body
    canvas.drawLine(Offset(cx, cy - 5), Offset(cx, cy + 6), strokePaint);
    // Arms
    canvas.drawLine(Offset(cx - 8, cy - 1), Offset(cx + 8, cy - 1), strokePaint);
    // Legs
    canvas.drawLine(Offset(cx, cy + 6), Offset(cx - 5, cy + 14), strokePaint);
    canvas.drawLine(Offset(cx, cy + 6), Offset(cx + 5, cy + 14), strokePaint);
  }

  // Type badge background
  final badgePaint = Paint()
    ..style = PaintingStyle.fill
    ..color = isSelected ? const Color(0xFFBF5534) : const Color(0xFF333333);
  final badge = _mannequinBadge(mannequin.mannequinType);
  final badgeRect = RRect.fromRectAndRadius(
    Rect.fromLTWH(cx + 8, cy - 18, badge.length * 6.5 + 4, 13),
    const Radius.circular(2),
  );
  canvas.drawRRect(badgeRect, badgePaint);
  _drawLabel(canvas, badge, badgeRect.center, size: 8, color: Colors.white);

  mannequinRects[mannequin.id] = Rect.fromCircle(center: Offset(cx, cy), radius: 18);
}
```
