## Drift companion syntax for nullable columns

After Agent 1's schema change, `zoneId` is nullable. In Drift's generated companion:
- For the **`insert()`** static constructor: nullable columns with no default become `Value<T?> colName = const Value.absent()`. Pass `zoneId: Value('z1')` for a valued row, or omit it for null.
- For the **`FixturesTableCompanion()`** default constructor: all fields use `Value<T>`. Use `zoneId: const Value(null)` for an explicitly-null zone_id.

Same applies to `planogramId`, `planogramIdBack` (nullable), and `wallAdjacent` (has default `false`).

    test('watchStoreLevelByStore returns only fixtures where zoneId IS NULL', () async {
      // Store-level fixture (no zone)
      await db.fixturesDao.upsert(FixturesTableCompanion.insert(
        id: 'sl1',

        fixtureType: 'partition',

      ));

      final storeLevelRows =
          await db.fixturesDao.watchStoreLevelByStore('store_a').first;
      expect(storeLevelRows.length, 1);
      expect(storeLevelRows.first.id, 'sl1');
      expect(storeLevelRows.first.zoneId, isNull);
    });

    test('new fields default correctly', () async {
      await db.fixturesDao.upsert(FixturesTableCompanion.insert(
        id: 'f_new',
        fixtureType: 'rack',
        updatedAt: DateTime(2025),
        storeId: const Value('store_a'),

      final rows = await db.fixturesDao.watchAll().first;
      final f = rows.first;
      expect(f.zoneId, isNull);
      expect(f.planogramId, isNull);
      expect(f.planogramIdBack, isNull);
      expect(f.wallAdjacent, isFalse);
    });

    test('planogramId and wallAdjacent round-trip correctly', () async {
      await db.fixturesDao.upsert(FixturesTableCompanion.insert(
        id: 'fp',
        fixtureType: 'partition',
        updatedAt: DateTime(2025),
        storeId: const Value('store_a'),
        planogramId: const Value('plano_1'),
        planogramIdBack: const Value('plano_2'),
        wallAdjacent: const Value(true),
      ));
      final rows = await db.fixturesDao.watchAll().first;
      final f = rows.first;
      expect(f.planogramId, 'plano_1');
      expect(f.planogramIdBack, 'plano_2');
      expect(f.wallAdjacent, isTrue);
    });

```dart
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:go_router/go_router.dart';
import 'package:merch_mobile/core/models/fixture.dart';
import 'package:merch_mobile/features/floor_builder/fixture_mini_panel.dart';

void main() {
  // Minimal GoRouter for context.push calls in FixtureMiniPanel.
  final router = GoRouter(
    routes: [
      GoRoute(
        path: '/',
        builder: (_, __) => const Scaffold(body: SizedBox.shrink()),
      ),
    ],
  );

  Widget buildPanel({
    required Fixture fixture,
    dynamic planogram,
    VoidCallback? onDismiss,
  }) {
    return MaterialApp.router(
      routerConfig: router,
      builder: (context, child) => Scaffold(
        body: FixtureMiniPanel(
          fixture: fixture,
          planogram: planogram,
          onDismiss: onDismiss ?? () {},
        ),
      ),
    );
  }

  final baseFixture = Fixture(
    id: 'f1',
    zoneId: 'z1',
    fixtureType: 'rack',
    widthFt: 4.0,
    depthFt: 2.0,
    label: 'WOMEN RACK',
    updatedAt: DateTime(2025),
  );

  testWidgets('shows fixture label and dimensions', (tester) async {
    await tester.pumpWidget(buildPanel(fixture: baseFixture));
    expect(find.text('WOMEN RACK'), findsOneWidget);
    expect(find.text('4.0 × 2.0 ft'), findsOneWidget);
  });

  testWidgets('shows No planogram when unassigned', (tester) async {
    await tester.pumpWidget(buildPanel(fixture: baseFixture, planogram: null));
    expect(find.text('No planogram'), findsOneWidget);
    expect(find.text('VIEW →'), findsNothing);
  });

  testWidgets('shows fixture type when label is empty', (tester) async {
    final noLabel = baseFixture.copyWith(label: '');
    await tester.pumpWidget(buildPanel(fixture: noLabel));
    expect(find.text('RACK'), findsOneWidget);
  });

  testWidgets('calls onDismiss when X tapped', (tester) async {
    var dismissed = false;
    await tester.pumpWidget(
      buildPanel(fixture: baseFixture, onDismiss: () => dismissed = true),
    );
    await tester.tap(find.byIcon(Icons.close));
    await tester.pump();
    expect(dismissed, isTrue);
  });
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:merch_mobile/core/database/app_database.dart';
import 'package:merch_mobile/features/floor_builder/planogram_picker_sheet.dart';

import '../../helpers/test_database.dart';

void main() {
  late AppDatabase db;
  setUp(() => db = createTestDatabase());
  tearDown(() async => await db.close());

  Future<List<PlanogramsTableData>> seedPlanograms() async {
    await db.planogramsDao.upsert(PlanogramsTableCompanion.insert(
      id: 'p1',
      title: 'Summer Collection',
      season: 'Summer 2025',
      updatedAt: DateTime(2025),
      storeId: const Value('s1'),
      fixtureId: const Value('f1'),
    ));
    await db.planogramsDao.upsert(PlanogramsTableCompanion.insert(
      id: 'p2',
      title: 'Winter Basics',
      season: 'Winter 2025',
      updatedAt: DateTime(2025),
      storeId: const Value('s1'),
      fixtureId: const Value('f1'),
    ));
    return db.planogramsDao.watchAll().first;
  }

  Widget buildSheet({
    required List<PlanogramsTableData> planograms,
    String? currentId,
    void Function(String?)? onSelect,
  }) {
    return MaterialApp(
      home: Scaffold(
        body: PlanogramPickerSheet(
          planograms: planograms,
          currentPlanogramId: currentId,
          fixtureLabel: 'WOMEN RACK',
          onSelect: onSelect ?? (_) {},
        ),
      ),
    );
  }

  testWidgets('shows all planogram titles', (tester) async {
    final planograms = await seedPlanograms();
    await tester.pumpWidget(buildSheet(planograms: planograms));
    await tester.pump();

    expect(find.text('Summer Collection'), findsOneWidget);
    expect(find.text('Winter Basics'), findsOneWidget);
  });

  testWidgets('search filters results', (tester) async {
    final planograms = await seedPlanograms();
    await tester.pumpWidget(buildSheet(planograms: planograms));
    await tester.pump();

    await tester.enterText(find.byType(TextField), 'summer');
    await tester.pump();

    expect(find.text('Summer Collection'), findsOneWidget);
    expect(find.text('Winter Basics'), findsNothing);
  });

  testWidgets('shows remove assignment when planogram is assigned', (tester) async {
    final planograms = await seedPlanograms();
    await tester.pumpWidget(buildSheet(planograms: planograms, currentId: 'p1'));
    await tester.pump();

    expect(find.text('Remove assignment'), findsOneWidget);
  });

  testWidgets('does not show remove assignment when unassigned', (tester) async {
    final planograms = await seedPlanograms();
    await tester.pumpWidget(buildSheet(planograms: planograms, currentId: null));
    await tester.pump();

    expect(find.text('Remove assignment'), findsNothing);
  });

  testWidgets('tapping remove calls onSelect with null', (tester) async {
    final planograms = await seedPlanograms();
    String? selected = 'p1';
    await tester.pumpWidget(buildSheet(
      planograms: planograms,
      currentId: 'p1',
      onSelect: (id) => selected = id,
    ));
    await tester.pump();

    await tester.tap(find.text('Remove assignment'));
    await tester.pumpAndSettle();

    expect(selected, isNull);
  });

  testWidgets('tapping a planogram tile calls onSelect with its id', (tester) async {
    final planograms = await seedPlanograms();
    String? selected;
    await tester.pumpWidget(buildSheet(
      planograms: planograms,
      onSelect: (id) => selected = id,
    ));
    await tester.pump();

    await tester.tap(find.text('Summer Collection'));
    await tester.pumpAndSettle();

    expect(selected, 'p1');
  });

  testWidgets('shows empty state when search yields no results', (tester) async {
    final planograms = await seedPlanograms();
    await tester.pumpWidget(buildSheet(planograms: planograms));
    await tester.pump();

    await tester.enterText(find.byType(TextField), 'zzznomatch');
    await tester.pump();

    expect(find.text('No planograms found'), findsOneWidget);
  });
}
```
